local function corner(inst, r)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, r or 8)
    c.Parent = inst
    return c
end
local function stroke(inst, col, thick)
    local s = Instance.new("UIStroke")
    s.Color = col or C.accent
    s.Thickness = thick or 1.5
    s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    s.Parent = inst
    return s
end
local function makeDraggable(frame, handle)
    handle = handle or frame
    local dragging, dragStart, startPos
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
end
local function gradient(inst, stops, rotation)
    local g = Instance.new("UIGradient")
    g.Color = ColorSequence.new(stops)
    g.Rotation = rotation or 0
    g.Parent = inst
    return g
end

            pcall(function() VirtualInputManager:SendKeyEvent(false, "W", false, game) end)
            farmSub.Text = "fly up · hold W · loop"
            setStatus("autofarm off", C.sub)
        end
    end)

    local foot = Instance.new("TextLabel")
    foot.Size = UDim2.new(1, 0, 0, 18)
    foot.Position = UDim2.new(0, 0, 1, -20)
    foot.BackgroundTransparency = 1
    foot.Text = BRAND .. " · v3"
    foot.Font = Enum.Font.Gotham
    foot.TextSize = 10
    foot.TextColor3 = Color3.fromRGB(70, 70, 80)
    foot.Parent = frame

    frame.BackgroundTransparency = 0
    topBar.BackgroundTransparency = 0
    fStroke.Transparency = 0
    TweenService:Create(frame, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
        Position = UDim2.new(0.5, -150, 0.5, -210),
    }):Play()

    return sg
end
