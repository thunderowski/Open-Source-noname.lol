local function shimmer(g, speed)
    speed = speed or 1.2
    task.spawn(function()
        while g.Parent do
            local t0 = tick()
            while tick() - t0 < speed and g.Parent do
                local a = (tick() - t0) / speed
                g.Offset = Vector2.new(a * 2.4 - 1.2, 0)
                task.wait(1/60)
            end
            task.wait(0.6)
        end
    end)
end

local function makeDraggable(frame, handle)
    handle = handle or frame
    local dragging, dragStart, startPos
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true; dragStart = input.Position; startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
end

            end
        end)
    end

    local function startFarm()
        local char = LP.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if not hrp or not hum then
            setStatus("no character", C.danger)
            return false
        end

        cleanupFarm()
        local startPos = hrp.Position
        local runId = state.farmRunId + 1
        state.farmRunId = runId

        local bv = Instance.new("BodyVelocity")
        bv.MaxForce = Vector3.new(1e6, 1e6, 1e6)
        bv.Velocity = Vector3.zero
        bv.Parent = hrp
        state.farmBV = bv
        hum.PlatformStand = true

        local phase = 1         -- 1 = rise, 2 = drive 4-7, 3 = hold W forever
        local altTarget = startPos + Vector3.new(0, FARM_ALT_HEIGHT, 0)
        local driveDist = math.random(FARM_DRIVE_MIN, FARM_DRIVE_MAX)
        local driveTarget = nil
        local wHeld = false

        local function holdW()
            if wHeld then return end
            wHeld = true
            pcall(function() hum:Move(Vector3.new(0, 0, -1), true) end)
            pcall(function() VirtualInputManager:SendKeyEvent(true, "W", false, game) end)
        end
        local function releaseW()
            if not wHeld then return end
            wHeld = false
            pcall(function() hum:Move(Vector3.zero, false) end)
            pcall(function() VirtualInputManager:SendKeyEvent(false, "W", false, game) end)
        end

        state.farmConn = RunService.Heartbeat:Connect(function()
            if state.farmRunId ~= runId then return end
            local c = LP.Character
            if not c then return end
            local h = c:FindFirstChild("HumanoidRootPart")
            if not h then return end

            if phase == 1 then
                -- ascend to altitude
                if h.Position.Y < altTarget.Y - 1 then
                    bv.Velocity = Vector3.new(0, FARM_RISE_SPEED, 0)
                else
                    bv.Velocity = Vector3.zero
                    local look = h.CFrame.LookVector
                    local flat = Vector3.new(look.X, 0, look.Z)
                    if flat.Magnitude < 0.01 then flat = Vector3.new(0, 0, -1) end
                    driveTarget = h.Position + flat.Unit * driveDist
                    phase = 2
                    holdW()
                end

            elseif phase == 2 then
                -- drive short distance with W held
                holdW()
                local delta = driveTarget - h.Position
                local flatDelta = Vector3.new(delta.X, 0, delta.Z)
                if flatDelta.Magnitude > 0.6 then
                    bv.Velocity = flatDelta.Unit * math.min(flatDelta.Magnitude * 4, FARM_DRIVE_SPEED)
                else
                    bv.Velocity = Vector3.zero
                    phase = 3
                    farmSub.Text = "holding W · " .. driveDist .. " studs done"
                    setStatus("autofarm holding W", C.accent)
                end

            elseif phase == 3 then
                -- forever: keep W held, keep driving forward at constant speed, no teleport
                holdW()
                local look = h.CFrame.LookVector
                local flat = Vector3.new(look.X, 0, look.Z)
                if flat.Magnitude < 0.01 then flat = Vector3.new(0, 0, -1) end
                bv.Velocity = flat.Unit * FARM_DRIVE_SPEED
            end
        end)
        return true
    end

    farmBtn.MouseButton1Click:Connect(function()
        state.farm = not state.farm
        farmSet(state.farm)
        if state.farm then
            if startFarm() then
                farmSub.Text = "running…"
                setStatus("autofarm on", C.accent)
            else
                state.farm = false
                farmSet(false)
            end
        else
            cleanupFarm()
            farmSub.Text = "fly up · drive · hold W forever"
            setStatus("autofarm off", C.sub)
        end
    end)

    TweenService:Create(frame, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
        Position = UDim2.new(0.5, -150, 0.5, -230),
    }):Play()

    return sg
end

    if started then return end
    started = true
    buildMain()
